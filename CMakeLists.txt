cmake_minimum_required(VERSION 3.20)
project(magicbox VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Installed privileged binaries resolve distribution libraries, never build-tree paths.
set(CMAKE_SKIP_RPATH TRUE)

find_package(Threads REQUIRED)
add_library(magicbox_core src/config.cpp src/engine.cpp)
target_include_directories(magicbox_core PUBLIC include)
target_compile_features(magicbox_core PUBLIC cxx_std_23)
target_link_libraries(magicbox_core PUBLIC Threads::Threads)
target_compile_options(magicbox_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)

find_path(DROGON_INCLUDE_DIR drogon/drogon.h REQUIRED)
find_path(JSONCPP_INCLUDE_DIR json/json.h PATH_SUFFIXES jsoncpp REQUIRED)
find_path(PCAP_INCLUDE_DIR pcap/pcap.h REQUIRED)
find_path(MNL_INCLUDE_DIR libmnl/libmnl.h REQUIRED)
find_path(SQLITE_INCLUDE_DIR sqlite3.h REQUIRED)
find_library(DROGON_LIBRARY drogon REQUIRED)
find_library(TRANTOR_LIBRARY trantor REQUIRED)
find_library(JSONCPP_LIBRARY NAMES jsoncpp libjsoncpp.so.26 REQUIRED)
find_library(PCAP_LIBRARY NAMES pcap libpcap.so.0.8 REQUIRED)
find_library(MNL_LIBRARY NAMES mnl libmnl.so.0 REQUIRED)
find_library(SQLITE_LIBRARY NAMES sqlite3 libsqlite3.so.0 REQUIRED)

add_library(magicbox_runtime src/json.cpp src/capture.cpp src/store.cpp src/process.cpp src/network.cpp)
target_include_directories(magicbox_runtime PUBLIC ${JSONCPP_INCLUDE_DIR} PRIVATE ${PCAP_INCLUDE_DIR} ${MNL_INCLUDE_DIR} ${SQLITE_INCLUDE_DIR})
target_link_libraries(magicbox_runtime PUBLIC magicbox_core ${JSONCPP_LIBRARY} PRIVATE ${PCAP_LIBRARY} ${MNL_LIBRARY} ${SQLITE_LIBRARY})
target_compile_options(magicbox_runtime PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(magicboxd src/main.cpp src/server.cpp)
target_include_directories(magicboxd SYSTEM PRIVATE ${DROGON_INCLUDE_DIR})
target_link_libraries(magicboxd PRIVATE magicbox_runtime ${DROGON_LIBRARY} ${TRANTOR_LIBRARY})
target_compile_options(magicboxd PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
include(CTest)
if(BUILD_TESTING)
  add_executable(magicbox_tests tests/core_tests.cpp)
  target_link_libraries(magicbox_tests PRIVATE magicbox_core)
  add_test(NAME core COMMAND magicbox_tests)
  set_tests_properties(core PROPERTIES TIMEOUT 10)
  add_executable(magicbox_runtime_tests tests/runtime_tests.cpp)
  target_link_libraries(magicbox_runtime_tests PRIVATE magicbox_runtime)
  add_test(NAME runtime COMMAND magicbox_runtime_tests)
  set_tests_properties(runtime PROPERTIES TIMEOUT 30)
endif()
install(TARGETS magicboxd RUNTIME DESTINATION bin)
install(DIRECTORY web/ DESTINATION share/magicbox/web)
